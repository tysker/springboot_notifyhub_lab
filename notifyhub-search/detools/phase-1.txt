# Create the notifications-v1 index with strict mapping
PUT notifications-v1
{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 0
  },
  "mappings": {
    "dynamic": "strict",
    "properties": {
      "id":         { "type": "keyword" },
      "channel":    { "type": "keyword" },
      "recipient":  { "type": "keyword" },
      "subject": {
        "type": "text",
        "fields": {
          "keyword": { "type": "keyword" }
        }
      },
      "message":    { "type": "text" },
      "status":     { "type": "keyword" },
      "retries":    { "type": "integer" },
      "durationMs": { "type": "long" },
      "createdAt":  { "type": "date" }
    }
  }
}

# Create a notification document

{
  "id": "<uuid>",
  "channel": "EMAIL",
  "recipient": "anna@example.com",

  "status": "SENT",
  "retries": 0,

# Create an alias for the notifications-v1 index
POST _aliases
{
  "actions": [
    {
      "add": {
        "index": "notifications-v1",
        "alias": "notifications",
        "is_write_index": true
      }
    }
  ]
}

# Update the status and retries of a notification document
POST notifications/_update/<uuid>
{
  "doc": {
    "status": "FAILED",
    "retries": 1
  }
}

# Delete a notification document
DELETE notifications/_doc/<uuid>
