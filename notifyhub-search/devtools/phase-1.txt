# Get information about the root of the OpenSearch cluster
GET /

# Get information about the OpenSearch cluster
GET _cat/health?v

# Get information about the nodes in the cluster
GET _cat/nodes?v

# Get information about the indices in the cluster
GET _cat/indices?v

# Get information about the aliases in the cluster
GET _cat/aliases?v
GET _alias/notifications
GET _cat/aliases/notifications?format=json

# Get information about all indices that start with "notifications-"
GET _cat/indices/notifications-?v

# Get information about the shards in the cluster
GET _cat/shards/notifications?v

# Get the mapping of the notifications-v1 index
GET notifications-v1/_mapping

# Get the document count of the notifications index
GET notifications/_count

# Analyze the text "Payment failed" using the subject field of the notifications index
GET notifications/_analyze
{
  "field": "subject",
  "text": "Payment failed"
}
GET notifications/_analyze
{
  "field": "subject.keyword",
  "text": "Payment failed"
}

# Re-index the notifications index after changing the mapping
POST notifications/_update_by_query

# SEARCH QUERIES

# Get the first 10 documents in the notifications index
GET notifications/_search
# By Size
GET notifications/_search?size=12

# Match
GET notifications/_search
{
  "query": {
    "match": {
      "subject": "payment FAILED"
    }
  }
}

# Term(s) GROUP BY
GET notifications/_search
{
  "size": 0,
  "aggs": {
    "per_status": {
      "terms": { "field": "status" }
    }
  }
}

# Term(s)
GET notifications/_search
GET notifications/_search
{
  "query": {
    "terms": {
        "status": ["FAILED", "SENT"]
    }
  }
}

# Term on text (no hits)
GET notifications/_search
{
  "query": {
    "term": {
      "subject": "Payment failed"
    }
  }
}

# Term on keyword (one hit)
GET notifications/_search
{
  "query": {
    "term": {
      "subject.keyword": "Payment failed"
    }
  }
}
# By default match uses or: any of the words is enough. With and, all of them must be present.
GET notifications/_search
{
  "query": {
    "match": {
      "subject": {
        "query": "payment FAILED",
        "operator": "and"
      }
    }
  }
}

# Create a test index with default mapping
PUT test-yellow

# Create the notifications-v1 index with strict mapping
PUT notifications-v1
{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 0
  },
  "mappings": {
    "dynamic": "strict",
    "properties": {
      "id":         { "type": "keyword" },
      "channel":    { "type": "keyword" },
      "recipient":  { "type": "keyword" },
      "subject": {
        "type": "text",
        "fields": {
          "keyword": { "type": "keyword" }
        }
      },
      "message":    { "type": "text" },
      "status":     { "type": "keyword" },
      "retries":    { "type": "integer" },
      "durationMs": { "type": "long" },
      "createdAt":  { "type": "date" }
    }
  }
}

# Create a notification document
PUT notifications/_doc/<uuid>
{
  "id": "<uuid>",
  "channel": "EMAIL",
  "recipient": "anna@example.com",
  "subject": "Payment received",
  "message": "We have received your payment of 250 DKK. Thank you!",
  "status": "SENT",
  "retries": 0,
  "durationMs": 120,
  "createdAt": "2026-10-17T10:00:00Z"
}

# Add a new field to an existing mapping
PUT notifications/_mapping
{
  "properties": {
    "template": { "type": "keyword" }
  }
}

# Add a sub-field to an existing field
PUT notifications/_mapping
{
  "properties": {
    "message": {
      "type": "text",
      "fields": {
        "keyword": { "type": "keyword" }
      }
    }
  }
}

# Create an alias for the notifications-v1 index
POST _aliases
{
  "actions": [
    {
      "add": {
        "index": "notifications-v1",
        "alias": "notifications",
        "is_write_index": true
      }
    }
  ]
}

# Update the status and retries of a notification document
POST notifications/_update/<uuid>
{
  "doc": {
    "status": "FAILED",
    "retries": 1
  }
}

# Delete a notification document
DELETE notifications/_doc/<uuid>
